Forge apt repository

Debian 12+ and Ubuntu 22.04+, amd64 and arm64.

sudo install -d -m 0755 /etc/apt/keyrings
curl -fsSL "$BASE/forge-archive-keyring.asc" | sudo tee /etc/apt/keyrings/forge.asc > /dev/null
echo "deb [signed-by=/etc/apt/keyrings/forge.asc] $BASE stable main" | sudo tee /etc/apt/sources.list.d/forge.list
sudo apt update && sudo apt install forge

Replace $BASE with the address of this page, without a trailing slash.
